// Função pura: step(sessao, mensagem, config) -> { sessao, respostas, eventos }
// mensagem: { tipo: 'text'|'image'|'document'|'other', texto, midiaId, arquivo }

const norm = (s) => (s || '').toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/\s+/g, ' ').trim();
const brl = (v) => 'R$ ' + v.toFixed(2).replace('.', ',');
const txt = (texto) => ({ tipo: 'text', texto });

const PEDE_HUMANO = /\b(atendente|humano|pessoa|falar com (alguem|voce|o dono)|reclama|problema|reembolso|devolu)/;
const RECOMECAR = /^(recomecar|reiniciar|comecar de novo|menu|inicio|cancelar)$/;
const RECUSA_SENSUAL = /(\b(sensual|nua|nude|nudes|lingerie|boudoir|sexy|ousad\w*|picante|adulto)\b|\+18|18\+)/;

function novaSessao(telefone) {
  return { telefone, etapa: 'inicio', tema: null, pacote: null, cenas: [], fotos: [], cenaPrevia: null, comprovante: null, humano: false, erros: 0, atualizado: Date.now() };
}

function menuTemas(cfg) {
  const linhas = cfg.temas.map((t, i) => `${i + 1}️⃣ ${t.nome}`).join('\n');
  return `Temas disponíveis 👇\n${linhas}\n\nÉ só responder com o número ou o nome do tema.`;
}

function listaPacotes(cfg) {
  return cfg.pacotes.map((p) => `✅ ${p.fotos} fotos por ${brl(p.preco)}`).join('\n');
}

function menuPacotes(cfg) {
  return `Qual pacote você prefere?\n${listaPacotes(cfg)}\n\nResponda com a quantidade de fotos 😊`;
}

function acharTema(texto, cfg) {
  const t = norm(texto);
  const n = t.match(/^\s*(\d{1,2})\s*$/) || (t.length <= 25 && (t.match(/\d+/g) || []).length === 1 ? t.match(/(\d{1,2})/) : null);
  if (n) { const i = parseInt(n[1], 10) - 1; return cfg.temas[i] || null; }
  let melhor = null;
  for (const tema of cfg.temas) {
    if (t.includes(norm(tema.nome))) return tema;
    for (const c of tema.chaves) if (t.includes(c)) melhor = melhor || tema;
  }
  if (!melhor && /aniversario|niver/.test(t)) return { ambiguo: 'aniversario' };
  if (!melhor && /corporativ|linkedin|profissional|curriculo/.test(t)) return { ambiguo: 'corporativo' };
  return melhor;
}

function acharPacote(texto, cfg) {
  const t = norm(texto).replace(',', '.');
  const preco = t.match(/(\d{1,2}\.\d{2})/);
  if (preco) { const p = cfg.pacotes.find((x) => Math.abs(x.preco - parseFloat(preco[1])) < 0.01); if (p) return p; }
  const nums = (t.match(/\d+/g) || []).map(Number);
  for (const n of nums) { const p = cfg.pacotes.find((x) => x.fotos === n); if (p) return p; }
  return null;
}

function acharCenas(texto, tema) {
  const t = norm(texto).toUpperCase();
  if (tema.codigos) {
    const cods = (t.match(/\b([A-Z])\s?0?(\d{1,2})\b/g) || []).map((c) => {
      const m = c.match(/([A-Z])\s?0?(\d{1,2})/); const n = parseInt(m[2], 10);
      return tema.codigos.includes(m[1]) && n >= 1 && n <= tema.cenas ? m[1] + String(n).padStart(2, '0') : null;
    }).filter(Boolean);
    return [...new Set(cods)];
  }
  const nums = (t.match(/\d{1,3}/g) || []).map(Number).filter((n) => n >= 1 && n <= tema.cenas);
  return [...new Set(nums)];
}

const faixa = (tema) => tema.dicaCodigos || `de 1 a ${tema.cenas}`;
const nomeCena = (c) => (typeof c === 'number' ? String(c).padStart(2, '0') : c);
const VOCE_ESCOLHE = /(voce escolhe|escolhe voce|vc escolhe|tanto faz|qualquer|pode escolher|nao sei|surpreenda)/;
const PRONTO = /^(pronto|pronta|so isso|e isso|terminei|acabei|ok|enviei|foi|mandei|ja mandei|ja enviei)\b/;

function quemFotos(s) {
  return s.tema === 'infantil' ? 'da criança' : s.tema === 'casal' || s.tema === 'natal' ? 'de cada pessoa que vai aparecer' : 'suas';
}

function pedidoResumo(s, cfg) {
  const tema = cfg.temas.find((t) => t.id === s.tema);
  return `${tema.nome} · cenas ${s.cenas.map(nomeCena).join(', ')} · ${s.pacote.fotos} fotos · Total ${brl(s.pacote.preco)}`;
}

// Perguntas frequentes: respondidas em qualquer etapa, depois o robô repete o que falta.
function faq(t, s, cfg) {
  const out = [];
  const add = (x) => { if (x && out.length < 2) out.push(x); };
  if (/previa|amostra|ver antes|ver primeiro|testar|teste antes|exemplo com (o )?meu/.test(t)) {
    if (s.previaEnviada) add('Sua prévia grátis já foi enviada aqui em cima 👆 Gostando, é só escolher o pacote para receber em alta qualidade e sem marca d\'água.');
    else add('Sim! A prévia é *grátis* 🎁 Você escolhe 1 cena, manda suas selfies e recebe uma amostra com o seu rosto (com marca d\'água). Só paga se gostar.');
  }
  if (/demora|prazo|quanto tempo|tempo de entrega|quando (fica|chega)/.test(t)) add(`A prévia chega em poucos minutos, e o ensaio completo em até ${cfg.prazoMin} minutos depois do Pix ⏱️`);
  if (/\b(quanto|valor|valores|preco|precos|custa|tabela)\b/.test(t)) add(`Nossos pacotes:\n${listaPacotes(cfg)}\n\nE a prévia é grátis: você só paga depois de ver a sua amostra 😊`);
  if (/como funciona|como e que funciona|como faco|como faz|explica/.test(t)) add(`Funciona assim:\n1️⃣ Você escolhe o tema e 1 cena do catálogo\n2️⃣ Manda de 3 a 6 selfies\n3️⃣ Recebe uma *prévia grátis* com o seu rosto\n4️⃣ Gostou? Escolhe o pacote, paga no Pix e recebe as fotos em até ${cfg.prazoMin} minutos 📸`);
  if (/golpe|confiavel|confianca|seguro|e real|e verdade|medo/.test(t)) add('Pode ficar tranquilo(a) 🙏 Você recebe a prévia grátis antes de pagar qualquer coisa, e se alguma foto não ficar parecida, a gente refaz.');
  if (/parecid|igual a mim|meu rosto|fica bom|fica real/.test(t)) add('A IA usa as suas selfies para manter o seu rosto 😊 Quanto mais nítidas e de frente, mais parecido fica. E você confere tudo na prévia grátis antes de pagar.');
  if (/\b(pix|pagar|pagamento|cartao)\b/.test(t) && !['pagamento', 'cenas'].includes(s.etapa)) add('O pagamento é por Pix, e só depois de você ver a prévia grátis e escolher o pacote 😊');
  return out.length ? out.join('\n\n') : null;
}

function lembrete(s, cfg) {
  const tema = cfg.temas.find((x) => x.id === s.tema);
  switch (s.etapa) {
    case 'tema': return menuTemas(cfg);
    case 'responsavel': return 'Você é o pai, a mãe ou o responsável pela criança? Responda *sim* ou *não*.';
    case 'cena_previa': return `Me diz o número da cena do catálogo que você mais gostou (${faixa(tema)}) para eu fazer sua prévia grátis. Se preferir, escreva *você escolhe*.`;
    case 'fotos_previa': return `Agora me mande de 3 a 6 fotos ${quemFotos(s)}, com o rosto bem visível. Quando terminar, escreva *pronto*.`;
    case 'aguardando_previa': return 'Sua prévia já está sendo criada 🎨 Chega aqui em poucos minutos.';
    case 'pacote': return menuPacotes(cfg);
    case 'cenas': return `Me manda mais ${s.pacote.fotos - s.cenas.length} número(s) de cena do catálogo (${faixa(tema)}).`;
    case 'pagamento': return `Seu pedido: ${pedidoResumo(s, cfg)}.\nAssim que pagar o Pix, me manda o print ou PDF do comprovante aqui.`;
    default: return null;
  }
}

function naoEntendi(s, R, E, texto, cfg) {
  s.erros = (s.erros || 0) + 1;
  if (s.erros >= 3) {
    s.humano = true; s.erros = 0;
    R.push(txt('Acho que não estou conseguindo te ajudar direito 😅 Vou chamar uma pessoa da equipe, que já te responde por aqui.'));
    E.push({ tipo: 'chamar_humano', telefone: s.telefone, motivo: 'robô não entendeu: ' + texto });
    return;
  }
  R.push(txt('Não entendi muito bem 🤔 ' + (lembrete(s, cfg) || '')));
}

function step(sessaoEntrada, msg, cfg = CONFIG_PADRAO) {
  const s = { ...sessaoEntrada, cenas: [...(sessaoEntrada.cenas || [])], fotos: [...(sessaoEntrada.fotos || [])] };
  s.atualizado = Date.now();
  const R = []; const E = [];
  const t = norm(msg.texto);
  const ok = () => { s.erros = 0; };

  // Atendimento humano em andamento: robô fica quieto.
  if (s.humano) return { sessao: s, respostas: R, eventos: [{ tipo: 'mensagem_para_humano', telefone: s.telefone, msg }] };

  if (msg.tipo === 'text' && PEDE_HUMANO.test(t)) {
    s.humano = true;
    R.push(txt('Certo! Vou chamar uma pessoa da nossa equipe para continuar com você. Já já te respondemos por aqui 🙂'));
    E.push({ tipo: 'chamar_humano', telefone: s.telefone, motivo: msg.texto });
    return { sessao: s, respostas: R, eventos: E };
  }

  if (msg.tipo === 'text' && RECOMECAR.test(t)) {
    const n = novaSessao(s.telefone); n.etapa = 'tema';
    R.push(txt('Sem problema, vamos recomeçar.\n\n' + menuTemas(cfg)));
    return { sessao: n, respostas: R, eventos: E };
  }

  if (msg.tipo === 'text' && RECUSA_SENSUAL.test(t) && ['inicio', 'tema'].includes(s.etapa)) {
    s.etapa = 'tema';
    R.push(txt('Esse tipo de ensaio a gente não faz 🙏 Mas temos vários temas lindos:\n\n' + menuTemas(cfg)));
    return { sessao: s, respostas: R, eventos: E };
  }

  // Perguntas no meio do caminho (não na primeira mensagem, que já ganha a explicação completa).
  if (msg.tipo === 'text' && s.etapa !== 'inicio' && !['producao', 'entregue'].includes(s.etapa)) {
    const resp = faq(t, s, cfg);
    const temOutraCoisa = s.etapa === 'tema' ? acharTema(msg.texto, cfg) : ['cena_previa', 'cenas'].includes(s.etapa) ? acharCenas(msg.texto, cfg.temas.find((x) => x.id === s.tema)).length : s.etapa === 'pacote' ? acharPacote(msg.texto, cfg) : null;
    if (resp && !temOutraCoisa) {
      ok();
      const l = lembrete(s, cfg);
      R.push(txt(resp + (l && s.etapa !== 'aguardando_previa' ? '\n\n' + l : '')));
      return { sessao: s, respostas: R, eventos: E };
    }
  }

  switch (s.etapa) {
    case 'inicio': {
      s.etapa = 'tema';
      const tema = msg.tipo === 'text' ? acharTema(msg.texto, cfg) : null;
      R.push(txt(`Oi! Seja bem-vindo(a) ao ${cfg.nome} 📸\n\nCriamos fotos de estúdio a partir das suas selfies.\n\nComo funciona:\n1️⃣ Você escolhe o tema e 1 cena\n2️⃣ Manda algumas selfies\n3️⃣ Recebe uma *prévia grátis* com o seu rosto 🎁\n4️⃣ Gostou? Escolhe o pacote (a partir de ${brl(cfg.pacotes[0].preco)}), paga no Pix e recebe tudo em até ${cfg.prazoMin} minutos`));
      if (tema && !tema.ambiguo) return escolherTema(s, tema, R, E, cfg);
      R.push(txt(menuTemas(cfg)));
      return { sessao: s, respostas: R, eventos: E };
    }

    case 'tema': {
      if (msg.tipo !== 'text') { R.push(txt('Recebi! Antes das fotos, me diz qual tema você quer 😊\n\n' + menuTemas(cfg))); break; }
      const tema = acharTema(msg.texto, cfg);
      if (tema && tema.ambiguo === 'aniversario') { ok(); R.push(txt('É aniversário feminino ou masculino? (Pode responder 1 ou 2)')); break; }
      if (tema && tema.ambiguo === 'corporativo') { ok(); R.push(txt('Corporativo masculino ou feminino? (Pode responder 3 ou 4)')); break; }
      if (!tema) { naoEntendi(s, R, E, msg.texto, cfg); break; }
      ok();
      return escolherTema(s, tema, R, E, cfg);
    }

    case 'responsavel': {
      if (/^(sim|s|sou|sou sim|confirmo|isso|ok|pode)\b/.test(t)) {
        ok(); s.responsavelOk = true; s.etapa = 'cena_previa';
        R.push(txt(lembrete(s, cfg)));
      } else if (/^(nao|n)\b/.test(t)) {
        s.etapa = 'tema';
        R.push(txt('Tudo bem. Para o ensaio infantil, as fotos e o pedido precisam vir do pai, da mãe ou do responsável pela criança. Se quiser outro tema:\n\n' + menuTemas(cfg)));
      } else {
        naoEntendi(s, R, E, msg.texto, cfg);
      }
      break;
    }

    case 'cena_previa': {
      const tema = cfg.temas.find((x) => x.id === s.tema);
      if (msg.tipo === 'image') { s.fotos.push(msg.midiaId); R.push(txt('Recebi sua foto ✅ ' + lembrete(s, cfg))); break; }
      let cena = null;
      if (VOCE_ESCOLHE.test(t)) cena = tema.codigos ? tema.codigos[0] + '01' : 1;
      else cena = acharCenas(msg.texto, tema)[0] || null;
      if (!cena) { naoEntendi(s, R, E, msg.texto, cfg); break; }
      ok(); s.cenaPrevia = cena; s.etapa = 'fotos_previa';
      if (s.fotos.length >= 3) return pedirPrevia(s, R, E, cfg);
      R.push(txt(`Perfeito, cena ${nomeCena(cena)} ✅\n\nAgora me mande de 3 a 6 fotos ${quemFotos(s)}: recentes, nítidas, com o rosto bem visível e sem filtro forte. Pode ser da galeria 😊\n\nQuando terminar, escreva *pronto*.`));
      break;
    }

    case 'fotos_previa': {
      if (msg.tipo === 'image') {
        ok(); s.fotos.push(msg.midiaId);
        if (s.fotos.length === 1) R.push(txt('Recebi! Pode mandar as outras. Quando terminar, escreva *pronto*.'));
        if (s.fotos.length >= 6) return pedirPrevia(s, R, E, cfg);
        break;
      }
      if (PRONTO.test(t)) {
        if (!s.fotos.length) { R.push(txt('Ainda não recebi nenhuma foto. Manda pelo menos 3, com o rosto bem visível 😊')); break; }
        if (s.fotos.length < 3 && !s.avisouPoucas) { s.avisouPoucas = true; R.push(txt(`Recebi ${s.fotos.length} ${s.fotos.length === 1 ? 'foto' : 'fotos'}. Com 3 ou mais fica bem mais parecido. Quer mandar mais? Se não, escreva *pronto* de novo.`)); break; }
        return pedirPrevia(s, R, E, cfg);
      }
      naoEntendi(s, R, E, msg.texto, cfg);
      break;
    }

    case 'aguardando_previa': {
      if (msg.tipo === 'image') { s.fotos.push(msg.midiaId); R.push(txt('Recebi mais uma foto ✅ Sua prévia já está sendo criada e chega em poucos minutos.')); break; }
      R.push(txt('Sua prévia está quase pronta 🎨 Chega aqui em poucos minutos.'));
      break;
    }

    case 'pacote': {
      const p = msg.tipo === 'text' ? acharPacote(msg.texto, cfg) : null;
      if (!p) {
        if (/^(sim|gostei|amei|quero|lindo|linda|perfeito|top|show|maravilh)/.test(t)) { ok(); R.push(txt('Que bom que gostou! 😍 ' + menuPacotes(cfg))); break; }
        naoEntendi(s, R, E, msg.texto, cfg); break;
      }
      ok(); s.pacote = p; s.cenas = s.cenaPrevia ? [s.cenaPrevia] : []; s.etapa = 'cenas';
      const tema = cfg.temas.find((x) => x.id === s.tema);
      const falta = p.fotos - s.cenas.length;
      R.push(txt(`Ótimo, pacote de ${p.fotos} fotos (${brl(p.preco)}).${s.cenaPrevia ? ` A cena ${nomeCena(s.cenaPrevia)} da prévia já está incluída ✅` : ''}\n\nAgora me manda ${falta === 1 ? 'o número de mais 1 cena' : `os números de mais ${falta} cenas`} do catálogo (${faixa(tema)}). Exemplo: ${exemploCenas(falta, tema)}${falta > 2 ? '\nSe preferir, escreva *você escolhe* que eu completo para você.' : ''}`));
      break;
    }

    case 'cenas': {
      const tema = cfg.temas.find((x) => x.id === s.tema);
      if (msg.tipo !== 'text') { R.push(txt(lembrete(s, cfg))); break; }
      const pac = acharPacote(msg.texto, cfg);
      if (/\b(pacote|fotos)\b/.test(t) && pac && pac.fotos !== s.pacote.fotos) {
        ok(); s.pacote = pac; s.cenas = s.cenaPrevia ? [s.cenaPrevia] : [];
        R.push(txt(`Trocado para o pacote de ${pac.fotos} fotos (${brl(pac.preco)}). ${lembrete(s, cfg)}`));
        break;
      }
      let novas = acharCenas(msg.texto, tema);
      if (!novas.length && VOCE_ESCOLHE.test(t)) novas = todasCenas(tema).filter((c) => !s.cenas.includes(c)).slice(0, s.pacote.fotos - s.cenas.length);
      if (!novas.length) { naoEntendi(s, R, E, msg.texto, cfg); break; }
      ok();
      s.cenas = [...new Set([...s.cenas, ...novas])];
      const falta = s.pacote.fotos - s.cenas.length;
      if (falta > 0) { R.push(txt(`Anotei as cenas ${s.cenas.map(nomeCena).join(', ')}. Falta${falta > 1 ? 'm' : ''} ${falta} para completar o pacote de ${s.pacote.fotos} fotos.`)); break; }
      if (falta < 0) {
        const maior = cfg.pacotes.find((x) => x.fotos >= s.cenas.length);
        R.push(txt(`Você escolheu ${s.cenas.length} cenas, mas o pacote é de ${s.pacote.fotos}.` + (maior ? ` Quer passar para o pacote de ${maior.fotos} fotos (${brl(maior.preco)})? Responda *${maior.fotos} fotos*, ou mande de novo as cenas.` : ' Mande de novo as cenas.')));
        s.cenas = s.cenaPrevia ? [s.cenaPrevia] : [];
        break;
      }
      s.etapa = 'pagamento';
      R.push(txt(`Confirmando seu pedido: ${pedidoResumo(s, cfg)}.\n\nPix: a chave vai na próxima mensagem, é só copiar 👇\nFavorecido: ${cfg.favorecido}\n\nDepois de pagar, me envie aqui o print ou o PDF do comprovante.`));
      R.push(txt(cfg.chavePix));
      E.push({ tipo: 'pedido_criado', telefone: s.telefone, resumo: pedidoResumo(s, cfg), valor: s.pacote.preco });
      break;
    }

    case 'pagamento': {
      if (msg.tipo === 'image' || msg.tipo === 'document') {
        ok(); s.comprovante = msg.midiaId;
        E.push({ tipo: 'comprovante_recebido', telefone: s.telefone, midiaId: msg.midiaId, resumo: pedidoResumo(s, cfg), valor: s.pacote.preco });
        if (s.fotos.length) {
          s.etapa = 'producao';
          R.push(txt(`Recebi o comprovante, obrigado! ✅\n\nSeu ensaio entrou em produção com as selfies que você já mandou. Você recebe as fotos aqui em até ${cfg.prazoMin} minutos 📸`));
          E.push({ tipo: 'pedido_pronto_para_producao', telefone: s.telefone, resumo: pedidoResumo(s, cfg), fotos: s.fotos, cenas: s.cenas, tema: s.tema });
        } else {
          s.etapa = 'fotos';
          R.push(txt(`Recebi o comprovante, obrigado! ✅\n\nAgora me envie de 3 a 6 fotos ${quemFotos(s)}: recentes, nítidas, com o rosto bem visível e sem filtro forte.\n\nQuando terminar de mandar, escreva *pronto*.`));
        }
      } else if (/(paguei|pago|feito|fiz o pix|ja paguei)/.test(t)) {
        R.push(txt('Perfeito! Me envia o print ou o PDF do comprovante aqui, por favor, para eu liberar a produção 🙏'));
      } else if (/(chave|copia|cola)/.test(t)) {
        R.push(txt(cfg.chavePix));
      } else {
        R.push(txt(`Seu pedido: ${pedidoResumo(s, cfg)}.\nAssim que pagar o Pix, me manda o comprovante aqui. Se quiser mudar algo, escreva *recomeçar*.`));
      }
      break;
    }

    case 'fotos': { // pedidos antigos (antes da prévia grátis)
      if (msg.tipo === 'image') {
        s.fotos.push(msg.midiaId);
        if (s.fotos.length === 1) R.push(txt('Recebi! Pode mandar as outras. Quando terminar, escreva *pronto*.'));
        if (s.fotos.length >= 6) return finalizar(s, R, E, cfg);
        break;
      }
      if (PRONTO.test(t)) {
        if (s.fotos.length < 1) { R.push(txt('Ainda não recebi nenhuma foto. Manda pelo menos 3, com o rosto bem visível 😊')); break; }
        if (s.fotos.length < 3 && !s.avisouPoucas) { s.avisouPoucas = true; R.push(txt(`Recebi ${s.fotos.length} ${s.fotos.length === 1 ? 'foto' : 'fotos'}. Com 3 ou mais o resultado fica bem mais parecido. Quer mandar mais? Se não, escreva *pronto* de novo.`)); break; }
        return finalizar(s, R, E, cfg);
      }
      R.push(txt('Me manda as fotos por aqui (como imagem). Quando terminar, escreva *pronto*.'));
      break;
    }

    case 'producao': {
      const simples = /^(ok|okay|blz|beleza|obrigad[oa]|valeu|show|top|certo|perfeito|otimo|aguardo|sim|ta bom|tudo bem)\b/.test(t);
      if (msg.tipo === 'text' && t && !simples) {
        s.humano = true;
        R.push(txt('Entendi! Vou passar sua mensagem para uma pessoa da equipe, que já te responde por aqui. Se for para mudar tema, cenas ou fotos, já pode dizer o que quer trocar 🙂'));
        E.push({ tipo: 'chamar_humano', telefone: s.telefone, motivo: 'durante a produção: ' + msg.texto });
        break;
      }
      if (msg.tipo === 'image') {
        s.humano = true;
        R.push(txt('Recebi a foto! Como seu ensaio já está em produção, vou passar para uma pessoa da equipe ver se dá para incluir 🙂'));
        E.push({ tipo: 'chamar_humano', telefone: s.telefone, motivo: 'enviou foto durante a produção' });
        break;
      }
      R.push(txt(`Seu ensaio já está em produção 🎨 Você recebe as fotos aqui em até ${cfg.prazoMin} minutos. Qualquer coisa, é só chamar.`));
      break;
    }

    case 'entregue': {
      if (/(refaz|refazer|nao ficou|nao parece|diferente|ruim)/.test(t)) {
        s.humano = true;
        R.push(txt('Poxa, sem problema! Me diz quais fotos não ficaram parecidas que a gente refaz 🙏 Uma pessoa da equipe já vai te responder.'));
        E.push({ tipo: 'chamar_humano', telefone: s.telefone, motivo: 'pedido de refação: ' + msg.texto });
      } else if (/(outro|mais|novo|de novo|quero mais)/.test(t)) {
        const n = novaSessao(s.telefone); n.etapa = 'tema';
        R.push(txt('Que bom que gostou! Vamos fazer outro 😍\n\n' + menuTemas(cfg)));
        return { sessao: n, respostas: R, eventos: E };
      } else {
        R.push(txt('Obrigado! 😊 Se quiser fazer outro ensaio, é só escrever *quero outro*. Se alguma foto precisar de ajuste, escreva *refazer*.'));
      }
      break;
    }

    default:
      s.etapa = 'tema';
      R.push(txt(menuTemas(cfg)));
  }
  return { sessao: s, respostas: R, eventos: E };
}

function todasCenas(tema) {
  if (tema.codigos) return tema.codigos.flatMap((p) => Array.from({ length: tema.cenas }, (_, i) => p + String(i + 1).padStart(2, '0')));
  return Array.from({ length: tema.cenas }, (_, i) => i + 1);
}

function escolherTema(s, tema, R, E, cfg) {
  s.tema = tema.id; s.cenas = []; s.pacote = null; s.cenaPrevia = null;
  R.push({ tipo: 'document', link: cfg.catalogoBase + tema.arquivo, arquivo: `Catálogo ${tema.nome}.pdf`, legenda: `Catálogo ${tema.nome} · cenas ${faixa(tema)}` });
  if (tema.responsavel) {
    s.etapa = 'responsavel';
    R.push(txt('Para o ensaio infantil, o pedido e as fotos precisam ser enviados pelo pai, pela mãe ou pelo responsável. Você é o responsável pela criança? Responda *sim* ou *não*.'));
  } else {
    s.etapa = 'cena_previa';
    R.push(txt(`Esse é o catálogo ${tema.nome} 👆\n\nEscolha *1 cena* para a sua *prévia grátis* 🎁 Me diz o número (${faixa(tema)}). Exemplo: ${exemploCenas(1, tema)}\n\nSe preferir, escreva *você escolhe*.`));
  }
  return { sessao: s, respostas: R, eventos: E };
}

function pedirPrevia(s, R, E, cfg) {
  s.etapa = 'aguardando_previa';
  R.push(txt(`Perfeito, recebi ${s.fotos.length} ${s.fotos.length === 1 ? 'foto' : 'fotos'} ✅\n\nEstou criando sua *prévia grátis* agora 🎨 Ela chega aqui em poucos minutos.`));
  E.push({ tipo: 'gerar_previa', telefone: s.telefone, tema: s.tema, cena: s.cenaPrevia, fotos: s.fotos });
  return { sessao: s, respostas: R, eventos: E };
}

function finalizar(s, R, E, cfg) {
  s.etapa = 'producao';
  R.push(txt(`Tudo certo! Recebi ${s.fotos.length} ${s.fotos.length === 1 ? 'foto' : 'fotos'} ✅\n\nSeu ensaio entrou em produção e você recebe as fotos aqui em até ${cfg.prazoMin} minutos 📸`));
  E.push({ tipo: 'pedido_pronto_para_producao', telefone: s.telefone, resumo: pedidoResumo(s, cfg), fotos: s.fotos, cenas: s.cenas, tema: s.tema });
  return { sessao: s, respostas: R, eventos: E };
}

function exemploCenas(n, tema) {
  const max = tema.cenas; const out = [];
  for (let i = 0; i < n; i++) {
    const v = Math.min(max, 3 + i * Math.max(1, Math.floor(max / (n + 1))));
    out.push(tema.codigos ? tema.codigos[i % tema.codigos.length] + String(v).padStart(2, '0') : v);
  }
  return out.join(', ');
}

// ===== Worker =====
// Worker Cloudflare do Estúdio Foto com IA: webhook da API oficial do WhatsApp + painel de pedidos.
// Variáveis (Settings > Variables): WA_TOKEN (secreto), WA_PHONE_ID, VERIFY_TOKEN, ADMIN_PASS (secreto), OWNER_EMAIL (opcional)
// Bindings: DB (D1). Opcional: EMAIL (send_email) para avisos ao dono.

const GRAPH = 'https://graph.facebook.com/v21.0';
const SCHEMA = [
  `CREATE TABLE IF NOT EXISTS sessoes (telefone TEXT PRIMARY KEY, dados TEXT NOT NULL, nome TEXT, atualizado INTEGER)`,
  `CREATE TABLE IF NOT EXISTS pedidos (id INTEGER PRIMARY KEY AUTOINCREMENT, telefone TEXT, nome TEXT, resumo TEXT, tema TEXT, cenas TEXT, valor REAL, status TEXT, comprovante TEXT, fotos TEXT, criado INTEGER, atualizado INTEGER)`,
  `CREATE TABLE IF NOT EXISTS mensagens (id TEXT PRIMARY KEY, telefone TEXT, direcao TEXT, tipo TEXT, texto TEXT, midia TEXT, ts INTEGER)`,
  `CREATE INDEX IF NOT EXISTS idx_msg_tel ON mensagens(telefone, ts)`,
  `CREATE TABLE IF NOT EXISTS geracoes (id INTEGER PRIMARY KEY AUTOINCREMENT, pedido INTEGER, tema TEXT, cena TEXT, status TEXT, midia TEXT, erro TEXT, tentativas INTEGER DEFAULT 0, atualizado INTEGER)`,
  `CREATE INDEX IF NOT EXISTS idx_ger_ped ON geracoes(pedido)`,
];

export default {
  async fetch(req, env, ctx) {
    const url = new URL(req.url);
    if (url.pathname === '/webhook' && req.method === 'GET') return verificar(url, env);
    if (url.pathname === '/webhook' && req.method === 'POST') {
      const corpo = await req.json().catch(() => null);
      ctx.waitUntil(processar(corpo, env).catch((e) => console.log('erro', e.stack)));
      return new Response('ok');
    }
    if (url.pathname.startsWith('/painel')) return painel(req, url, env);
    return new Response('Estúdio Foto com IA', { status: 200 });
  },
  async scheduled(event, env, ctx) {
    ctx.waitUntil(processarFila(env).catch((e) => console.log('fila erro', e.stack)));
  },
};

function verificar(url, env) {
  const ok = url.searchParams.get('hub.mode') === 'subscribe' && url.searchParams.get('hub.verify_token') === env.VERIFY_TOKEN;
  return ok ? new Response(url.searchParams.get('hub.challenge')) : new Response('proibido', { status: 403 });
}

let schemaOk = false;
async function garantirSchema(env) {
  if (schemaOk) return;
  for (const q of SCHEMA) await env.DB.prepare(q).run();
  try { await env.DB.prepare('ALTER TABLE geracoes ADD COLUMN tipo TEXT').run(); } catch (e) { /* já existe */ }
  schemaOk = true;
}

async function processar(corpo, env) {
  await garantirSchema(env);
  for (const entry of corpo?.entry || []) for (const ch of entry.changes || []) {
    const v = ch.value || {};
    const nomes = Object.fromEntries((v.contacts || []).map((c) => [c.wa_id, c.profile?.name || '']));
    for (const m of v.messages || []) {
      const novo = await env.DB.prepare('INSERT OR IGNORE INTO mensagens (id, telefone, direcao, tipo, texto, midia, ts) VALUES (?,?,?,?,?,?,?)')
        .bind(m.id, m.from, 'in', m.type, m.text?.body || m.image?.caption || m.document?.filename || '', m.image?.id || m.document?.id || null, Number(m.timestamp) * 1000).run();
      if (!novo.meta.changes) continue; // repetição do webhook
      await tratarMensagem(m, nomes[m.from] || '', env);
    }
  }
}

function converter(m) {
  if (m.type === 'text') return { tipo: 'text', texto: m.text.body };
  if (m.type === 'image') return { tipo: 'image', midiaId: m.image.id, texto: m.image.caption || '' };
  if (m.type === 'document') return { tipo: 'document', midiaId: m.document.id, arquivo: m.document.filename, texto: m.document.caption || '' };
  if (m.type === 'button') return { tipo: 'text', texto: m.button.text };
  if (m.type === 'interactive') return { tipo: 'text', texto: m.interactive.button_reply?.title || m.interactive.list_reply?.title || '' };
  if (m.type === 'audio') return { tipo: 'other', texto: '' };
  return { tipo: 'other', texto: '' };
}

async function carregarSessao(env, tel) {
  const r = await env.DB.prepare('SELECT dados FROM sessoes WHERE telefone=?').bind(tel).first();
  return r ? JSON.parse(r.dados) : novaSessao(tel);
}
async function salvarSessao(env, s, nome) {
  await env.DB.prepare('INSERT INTO sessoes (telefone, dados, nome, atualizado) VALUES (?,?,?,?) ON CONFLICT(telefone) DO UPDATE SET dados=excluded.dados, nome=COALESCE(NULLIF(excluded.nome,\'\'), sessoes.nome), atualizado=excluded.atualizado')
    .bind(s.telefone, JSON.stringify(s), nome, Date.now()).run();
}

async function tratarMensagem(m, nome, env) {
  const tel = m.from;
  let s = await carregarSessao(env, tel);
  const msg = converter(m);
  if (msg.tipo === 'other' && !s.humano) {
    await enviar(env, tel, { tipo: 'text', texto: 'Por aqui eu só consigo ler mensagens de texto, fotos e PDF 🙏 Pode me escrever?' });
    return;
  }
  const r = step(s, msg, CONFIG_PADRAO);
  await salvarSessao(env, r.sessao, nome);
  for (const resp of r.respostas) {
    await enviar(env, tel, resp);
    if (resp.tipo === 'document') await new Promise((ok) => setTimeout(ok, 3500));
  }
  for (const ev of r.eventos) await registrarEvento(env, ev, r.sessao, nome);
}

async function registrarEvento(env, ev, s, nome) {
  const agora = Date.now();
  if (ev.tipo === 'gerar_previa') {
    const tema = CONFIG_PADRAO.temas.find((t) => t.id === ev.tema);
    const resumo = `Prévia grátis · ${tema ? tema.nome : ev.tema} · cena ${codCena(ev.tema, ev.cena)}`;
    if (!s.pedidoId) {
      const res = await env.DB.prepare('INSERT INTO pedidos (telefone, nome, resumo, tema, cenas, valor, status, fotos, criado, atualizado) VALUES (?,?,?,?,?,?,?,?,?,?)')
        .bind(s.telefone, nome, resumo, ev.tema, JSON.stringify([ev.cena]), 0, 'previa', JSON.stringify(ev.fotos), agora, agora).run();
      s.pedidoId = res.meta.last_row_id; await salvarSessao(env, s, nome);
    } else {
      await env.DB.prepare('UPDATE pedidos SET resumo=?, tema=?, cenas=?, status=?, fotos=?, atualizado=? WHERE id=?').bind(resumo, ev.tema, JSON.stringify([ev.cena]), 'previa', JSON.stringify(ev.fotos), agora, s.pedidoId).run();
      await env.DB.prepare("DELETE FROM geracoes WHERE pedido=? AND tipo='previa'").bind(s.pedidoId).run();
    }
    if (env.OPENAI_API_KEY) await env.DB.prepare('INSERT INTO geracoes (pedido, tema, cena, status, tentativas, atualizado, tipo) VALUES (?,?,?,?,0,?,?)').bind(s.pedidoId, ev.tema, codCena(ev.tema, ev.cena), 'pendente', agora, 'previa').run();
    else await avisarDono(env, `Prévia pedida: ${nome || s.telefone}`, `${resumo}\nAbra o painel: /painel?c=${s.telefone}`);
  }
  if (ev.tipo === 'pedido_criado' && s.pedidoId) {
    await env.DB.prepare('UPDATE pedidos SET resumo=?, tema=?, cenas=?, valor=?, status=?, atualizado=? WHERE id=?').bind(ev.resumo, s.tema, JSON.stringify(s.cenas), ev.valor, 'aguardando_pagamento', agora, s.pedidoId).run();
  } else if (ev.tipo === 'pedido_criado') {
    const res = await env.DB.prepare('INSERT INTO pedidos (telefone, nome, resumo, tema, cenas, valor, status, fotos, criado, atualizado) VALUES (?,?,?,?,?,?,?,?,?,?)')
      .bind(s.telefone, nome, ev.resumo, s.tema, JSON.stringify(s.cenas), ev.valor, 'aguardando_pagamento', '[]', agora, agora).run();
    s.pedidoId = res.meta.last_row_id; await salvarSessao(env, s, nome);
  }
  if (ev.tipo === 'comprovante_recebido' && s.pedidoId) {
    await env.DB.prepare('UPDATE pedidos SET status=?, comprovante=?, atualizado=? WHERE id=?').bind('comprovante_recebido', ev.midiaId, agora, s.pedidoId).run();
  }
  if (ev.tipo === 'pedido_pronto_para_producao' && s.pedidoId) {
    const auto = !!env.OPENAI_API_KEY;
    await env.DB.prepare('UPDATE pedidos SET status=?, fotos=?, atualizado=? WHERE id=?').bind(auto ? 'gerando' : 'produzir', JSON.stringify(ev.fotos), agora, s.pedidoId).run();
    if (auto) {
      await env.DB.prepare("DELETE FROM geracoes WHERE pedido=? AND tipo='previa'").bind(s.pedidoId).run();
      await criarGeracoes(env, s.pedidoId, ev.tema, ev.cenas);
    }
    else await avisarDono(env, `Pedido #${s.pedidoId} pronto para produzir`, `${nome || s.telefone}\n${ev.resumo}\n${ev.fotos.length} fotos de referência.\nAbra o painel: /painel`);
  }
  if (ev.tipo === 'chamar_humano') await avisarDono(env, `Cliente pediu atendimento: ${nome || s.telefone}`, `Mensagem: ${ev.motivo}\nAbra o painel: /painel`);
}

async function avisarDono(env, assunto, texto) {
  if (!env.EMAIL || !env.OWNER_EMAIL) return;
  const remetente = env.FROM_EMAIL || 'robo@austriacoexplica.com.br';
  const raw = `From: Estudio Foto com IA <${remetente}>\r\nTo: ${env.OWNER_EMAIL}\r\nSubject: =?UTF-8?B?${b64(assunto)}?=\r\nMIME-Version: 1.0\r\nContent-Type: text/plain; charset=UTF-8\r\nContent-Transfer-Encoding: base64\r\n\r\n${b64(texto)}`;
  try {
    const { EmailMessage } = await import('cloudflare:email');
    await env.EMAIL.send(new EmailMessage(remetente, env.OWNER_EMAIL, raw));
  } catch (e) { console.log('email falhou', e.message); }
}
const b64 = (s) => btoa(String.fromCharCode(...new TextEncoder().encode(s)));

async function graph(env, caminho, init = {}) {
  const r = await fetch(GRAPH + caminho, { ...init, headers: { Authorization: 'Bearer ' + env.WA_TOKEN, ...(init.headers || {}) } });
  const j = await r.json().catch(() => ({}));
  if (!r.ok) { console.log('graph erro', caminho, JSON.stringify(j)); try { await env.DB.prepare('INSERT INTO mensagens (id, telefone, direcao, tipo, texto, midia, ts) VALUES (?,?,?,?,?,?,?)').bind('err-' + crypto.randomUUID(), '', 'erro', 'graph', JSON.stringify(j).slice(0, 1000), caminho, Date.now()).run(); } catch (e) {} }
  return j;
}

const waTo = (tel) => (/^55\d{2}[6-9]\d{7}$/.test(tel) ? tel.slice(0, 4) + '9' + tel.slice(4) : tel);
async function enviar(env, tel, resp) {
  let corpo;
  if (resp.tipo === 'text') corpo = { type: 'text', text: { body: resp.texto, preview_url: true } };
  else if (resp.tipo === 'document') corpo = { type: 'document', document: { link: resp.link, filename: resp.arquivo, caption: resp.legenda } };
  else if (resp.tipo === 'image') corpo = { type: 'image', image: resp.midiaId ? { id: resp.midiaId, caption: resp.legenda } : { link: resp.link, caption: resp.legenda } };
  const j = await graph(env, `/${env.WA_PHONE_ID}/messages`, {
    method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ messaging_product: 'whatsapp', to: waTo(tel), ...corpo }),
  });
  const id = j.messages?.[0]?.id || 'out-' + crypto.randomUUID();
  await env.DB.prepare('INSERT OR IGNORE INTO mensagens (id, telefone, direcao, tipo, texto, midia, ts) VALUES (?,?,?,?,?,?,?)')
    .bind(id, tel, 'out', resp.tipo, resp.texto || resp.legenda || resp.arquivo || '', resp.midiaId || resp.link || null, Date.now()).run();
  return j;
}

// ---------------- Painel ----------------
function autorizado(req, env) {
  const h = req.headers.get('Authorization') || '';
  if (!h.startsWith('Basic ')) return false;
  const [, senha] = atob(h.slice(6)).split(':');
  return senha && senha === env.ADMIN_PASS;
}

async function painel(req, url, env) {
  if (!autorizado(req, env)) return new Response('Senha necessária', { status: 401, headers: { 'WWW-Authenticate': 'Basic realm="Painel"' } });
  await garantirSchema(env);
  const p = url.pathname;
  if (p === '/painel/diag') {
    const info = await graph(env, '/' + env.WA_PHONE_ID + '?fields=display_phone_number,verified_name,quality_rating');
    return new Response(JSON.stringify({ phone: env.WA_PHONE_ID, tokenLen: (env.WA_TOKEN || '').length, tokenIni: (env.WA_TOKEN || '').slice(0, 3), info }, null, 2), { headers: { 'Content-Type': 'application/json' } });
  }
  if (p === '/painel/midia') {
    const info = await graph(env, '/' + url.searchParams.get('id'));
    if (!info.url) return new Response('mídia não encontrada', { status: 404 });
    const r = await fetch(info.url, { headers: { Authorization: 'Bearer ' + env.WA_TOKEN } });
    return new Response(r.body, { headers: { 'Content-Type': info.mime_type || 'application/octet-stream', 'Cache-Control': 'private, max-age=3600' } });
  }
  if (p === '/painel/acao' && req.method === 'POST') {
    const f = await req.formData();
    const id = Number(f.get('pedido')); const tel = f.get('telefone');
    const acao = f.get('acao');
    if (acao === 'enviar_fotos') {
      const arquivos = f.getAll('fotos').filter((x) => x && x.size);
      for (const a of arquivos) {
        const fd = new FormData(); fd.append('messaging_product', 'whatsapp'); fd.append('type', a.type || 'image/jpeg'); fd.append('file', a, a.name);
        const up = await graph(env, `/${env.WA_PHONE_ID}/media`, { method: 'POST', body: fd });
        if (up.id) await enviar(env, tel, { tipo: 'image', midiaId: up.id });
      }
      if (f.get('finalizar')) await finalizarPedido(env, id, tel);
    }
    if (acao === 'finalizar') await finalizarPedido(env, id, tel);
    if (acao === 'mensagem') {
      const texto = (f.get('texto') || '').trim();
      if (texto) await enviar(env, tel, { tipo: 'text', texto });
    }
    if (acao === 'assumir' || acao === 'devolver') {
      const s = await carregarSessao(env, tel); s.humano = acao === 'assumir'; await salvarSessao(env, s, '');
    }
    if (acao === 'refazer') {
      await env.DB.prepare("UPDATE geracoes SET status='pendente', erro=NULL, tentativas=0, atualizado=? WHERE id=? AND pedido=?").bind(Date.now(), Number(f.get('ger')), id).run();
      const gr = await env.DB.prepare('SELECT tipo FROM geracoes WHERE id=?').bind(Number(f.get('ger'))).first();
      if (gr && gr.tipo === 'previa') {
        await env.DB.prepare("UPDATE pedidos SET status='previa', atualizado=? WHERE id=?").bind(Date.now(), id).run();
        const ss = await carregarSessao(env, tel); ss.etapa = 'aguardando_previa'; ss.humano = false; ss.previaEnviada = false; await salvarSessao(env, ss, '');
      } else await env.DB.prepare("UPDATE pedidos SET status='gerando', atualizado=? WHERE id=?").bind(Date.now(), id).run();
    }
    if (acao === 'refazer_erros') {
      await env.DB.prepare("UPDATE geracoes SET status='pendente', erro=NULL, tentativas=0, atualizado=? WHERE pedido=? AND status='erro'").bind(Date.now(), id).run();
      await env.DB.prepare("UPDATE pedidos SET status='gerando', atualizado=? WHERE id=?").bind(Date.now(), id).run();
    }
    if (acao === 'aprovar') await aprovarEnviar(env, id, tel);
    if (acao === 'gerar_ia') {
      const p = await env.DB.prepare('SELECT * FROM pedidos WHERE id=?').bind(id).first();
      const ja = await env.DB.prepare('SELECT COUNT(*) n FROM geracoes WHERE pedido=?').bind(id).first();

async function baixarMidiaWA(env, id) {
  const info = await graph(env, '/' + id);
  if (!info.url) throw new Error('referência indisponível no WhatsApp');
  const r = await fetch(info.url, { headers: { Authorization: 'Bearer ' + env.WA_TOKEN } });
  if (!r.ok) throw new Error('falha ao baixar referência ' + r.status);
  return new Blob([await r.arrayBuffer()], { type: info.mime_type || 'image/jpeg' });
}

function b64ParaBytes(b64) {
  const bin = atob(b64); const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

async function gerarImagem(env, tema, cena, refs, previa = false) {
  const tpl = await fetch(SITE_CENAS + pastaTema(tema) + '/' + cena + '.jpg');
  if (!tpl.ok) throw new Error('cena do catálogo não encontrada: ' + cena);
  const template = new Blob([await tpl.arrayBuffer()], { type: 'image/jpeg' });
  const montar = (extra) => {
    const fd = new FormData();
    fd.append('model', env.OPENAI_IMAGE_MODEL || 'gpt-image-2.5-sunburst');
    fd.append('prompt', previa ? promptCena(tema).replace('no text, no logos, no watermark', 'no logos') + `
THIS IS A FREE PREVIEW: overlay a large semi-transparent white diagonal watermark reading "PRÉVIA" two or three times across the image (about 35% opacity), clearly visible but never covering the eyes.` : promptCena(tema));
    fd.append('size', env.OPENAI_IMAGE_SIZE || '1024x1536');
    fd.append('quality', previa ? (env.OPENAI_PREVIA_QUALITY || 'medium') : (env.OPENAI_IMAGE_QUALITY || 'high'));
    fd.append('n', '1');
    for (const [k, v] of Object.entries(extra)) fd.append(k, v);
    refs.forEach((b, i) => fd.append('image[]', b, `cliente-${i + 1}.jpg`));
    fd.append('image[]', template, 'cena.jpg');
    return fd;
  };
  const chamar = (fd) => fetch('https://api.openai.com/v1/images/edits', { method: 'POST', headers: { Authorization: 'Bearer ' + env.OPENAI_API_KEY }, body: fd });
  let r = await chamar(montar({ output_format: 'jpeg', output_compression: previa ? '60' : '85' }));
  let j = await r.json().catch(() => ({}));
  if (!r.ok && /output_(format|compression)/i.test(j?.error?.message || '')) { r = await chamar(montar({})); j = await r.json().catch(() => ({})); }
  if (!r.ok) throw new Error('OpenAI: ' + (j?.error?.message || r.status));
  const b64 = j.data?.[0]?.b64_json;
  if (!b64) throw new Error('OpenAI não devolveu imagem');
  const mime = /^\/9j/.test(b64) ? 'image/jpeg' : 'image/png';
  const fd = new FormData();
  fd.append('messaging_product', 'whatsapp'); fd.append('type', mime);
  fd.append('file', new Blob([b64ParaBytes(b64)], { type: mime }), 'foto.' + (mime === 'image/png' ? 'png' : 'jpg'));
  const up = await graph(env, `/${env.WA_PHONE_ID}/media`, { method: 'POST', body: fd });
  if (!up.id) throw new Error('falha ao guardar a foto gerada');
  return up.id;
}

async function processarFila(env) {
  await garantirSchema(env);
  if (!env.OPENAI_API_KEY) return;
  const inicio = Date.now(); const LIMITE = 11 * 60e3; const PARALELO = Number(env.OPENAI_PARALELO || 3);
  const refsCache = new Map();
  const pegarJob = async () => {
    const j = await env.DB.prepare("SELECT * FROM geracoes WHERE status='pendente' OR (status='gerando' AND atualizado < ?) ORDER BY CASE WHEN tipo='previa' THEN 0 ELSE 1 END, id LIMIT 1").bind(Date.now() - 8 * 60e3).first();
    if (!j) return null;
    const ok = await env.DB.prepare("UPDATE geracoes SET status='gerando', atualizado=? WHERE id=? AND (status='pendente' OR (status='gerando' AND atualizado < ?))").bind(Date.now(), j.id, Date.now() - 8 * 60e3).run();
    return ok.meta.changes ? j : pegarJob();
  };
  const trabalhador = async () => {
    while (Date.now() - inicio < LIMITE) {
      const job = await pegarJob(); if (!job) return;
      try {
        const ped = await env.DB.prepare('SELECT * FROM pedidos WHERE id=?').bind(job.pedido).first();
        if (!ped || ped.status === 'cancelado') { await env.DB.prepare("UPDATE geracoes SET status='cancelado' WHERE id=?").bind(job.id).run(); continue; }
        if (!refsCache.has(ped.id)) refsCache.set(ped.id, Promise.all(JSON.parse(ped.fotos || '[]').slice(0, 4).map((id) => baixarMidiaWA(env, id))));
        const refs = await refsCache.get(ped.id);
        const midia = await gerarImagem(env, job.tema, job.cena, refs, job.tipo === 'previa');
        await env.DB.prepare("UPDATE geracoes SET status='ok', midia=?, erro=NULL, atualizado=? WHERE id=?").bind(midia, Date.now(), job.id).run();
      } catch (e) {
        const tent = (job.tentativas || 0) + 1;
        await env.DB.prepare('UPDATE geracoes SET status=?, erro=?, tentativas=?, atualizado=? WHERE id=?').bind(tent >= 2 ? 'erro' : 'pendente', String(e.message || e).slice(0, 400), tent, Date.now(), job.id).run();
      }
      if (job.tipo === 'previa') await previaPronta(env, job.id);
      else await verificarPedido(env, job.pedido);
    }
  };
  await Promise.all(Array.from({ length: PARALELO }, trabalhador));
}

  if (!ped || ped.status !== 'previa') return;
  const tel = ped.telefone; const s = await carregarSessao(env, tel);
  if (g.status === 'ok') {
    await env.DB.prepare("UPDATE pedidos SET status='previa_enviada', atualizado=? WHERE id=?").bind(Date.now(), ped.id).run();
    if (s.etapa !== 'aguardando_previa') return;

    s.previaEnviada = true; s.etapa = 'pacote'; await salvarSessao(env, s, '');

  } else {
    await env.DB.prepare("UPDATE pedidos SET status='produzir', atualizado=? WHERE id=?").bind(Date.now(), ped.id).run();
    s.humano = true; await salvarSessao(env, s, '');

async function verificarPedido(env, pedido) {
  const r = (await env.DB.prepare('SELECT status, COUNT(*) n FROM geracoes WHERE pedido=? GROUP BY status').bind(pedido).all()).results;
  const c = Object.fromEntries(r.map((x) => [x.status, x.n]));
  if (c.pendente || c.gerando) return;
  const ped = await env.DB.prepare('SELECT * FROM pedidos WHERE id=?').bind(pedido).first();
  if (!ped || ped.status !== 'gerando') return;
  if (env.AUTO_ENVIO === 'sim' && !c.erro) return aprovarEnviar(env, pedido, ped.telefone);
  await env.DB.prepare("UPDATE pedidos SET status='revisar', atualizado=? WHERE id=? AND status='gerando'").bind(Date.now(), pedido).run();
  await avisarDono(env, `Pedido #${pedido} pronto para revisar`, `${ped.nome || ped.telefone}\n${ped.resumo}\n${c.ok || 0} fotos geradas${c.erro ? `, ${c.erro} com erro` : ''}.\nAbra o painel: /painel?c=${ped.telefone}`);
}

async function aprovarEnviar(env, pedido, tel) {
  const fotos = (await env.DB.prepare("SELECT * FROM geracoes WHERE pedido=? AND status='ok' ORDER BY id").bind(pedido).all()).results;
  for (const g of fotos) await enviar(env, tel, { tipo: 'image', midiaId: g.midia, legenda: `Cena ${g.cena}` });
  if (fotos.length) await finalizarPedido(env, pedido, tel);
}
